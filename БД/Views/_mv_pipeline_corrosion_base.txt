-- Исключаем выбросы: если в одной точке 2 замера с разницей < 30 дней,
-- оставляем только последний. LAG считается по уже отфильтрованной последовательности.

DROP MATERIALIZED VIEW IF EXISTS public._mv_pipeline_corrosion_base;

CREATE MATERIALIZED VIEW public._mv_pipeline_corrosion_base
TABLESPACE pg_default
AS
WITH raw_base AS (
    SELECT
        nm.id AS measurement_id,
        ni.id AS installation_id,
        ni.name AS installation,
        ne.id AS equipment_id,
        ne.name AS equipment,
        ne.start_date_of_operation,
        nc.id AS component_id,
        nc.name AS component,
        nm.contour,
        nm.mms,
        nm.measurement_date AS curr_measurement_date,
        nm.measurement AS curr_thickness,
        nm.nominal_thickness_mmc,
        nm.tmin_mmc,
        nc.initial_thickness,
        nc.component_type,
        nmm.code AS measurement_method_code,
        LEAD(nm.measurement_date) OVER w AS next_measurement_date
    FROM measurements nm
    JOIN components nc ON nc.id = nm.component_id
    JOIN equipment ne ON ne.id = nc.equipment_id
    JOIN installations ni ON ni.id = ne.installation_id
    LEFT JOIN measurement_methods nmm ON nmm.id = nm.measurement_method_id
    WHERE nc.component_type = 'Участок трубопровода'
    WINDOW w AS (PARTITION BY ne.id, nc.id, nm.mms ORDER BY nm.measurement_date, nm.id)
),
filtered_base AS (
    -- Исключаем замеры, если следующий замер в той же точке — менее чем через 30 дней
    -- (оставляем последний в паре близких замеров)
    SELECT
        measurement_id,
        installation_id,
        installation,
        equipment_id,
        equipment,
        start_date_of_operation,
        component_id,
        component,
        contour,
        mms,
        curr_measurement_date,
        curr_thickness,
        nominal_thickness_mmc,
        tmin_mmc,
        initial_thickness,
        component_type,
        measurement_method_code
    FROM raw_base
    WHERE next_measurement_date IS NULL
       OR (next_measurement_date - curr_measurement_date) >= 30
),
base AS (
    SELECT
        f.measurement_id,
        f.installation_id,
        f.installation,
        f.equipment_id,
        f.equipment,
        f.start_date_of_operation,
        f.component_id,
        f.component,
        f.contour,
        f.mms,
        f.curr_measurement_date,
        f.curr_thickness,
        LAG(f.curr_measurement_date) OVER w AS prev_measurement_date,
        LAG(f.curr_thickness) OVER w AS prev_thickness,
        LAG(f.curr_measurement_date, 2) OVER w AS prev2_measurement_date,
        LAG(f.curr_thickness, 2) OVER w AS prev2_thickness,
        f.nominal_thickness_mmc,
        f.tmin_mmc,
        f.initial_thickness,
        f.component_type,
        f.measurement_method_code
    FROM filtered_base f
    WINDOW w AS (PARTITION BY f.equipment_id, f.component_id, f.mms ORDER BY f.curr_measurement_date, f.measurement_id)
),
derived AS (
    SELECT
        b.*,
        COALESCE(b.nominal_thickness_mmc, b.initial_thickness) AS nominal_eff,
        b.tmin_mmc AS tmin_eff,
        EXTRACT(epoch FROM (b.curr_measurement_date::timestamp - b.prev_measurement_date::timestamp)) / 31557600.0 AS delta_years,
        EXTRACT(epoch FROM (b.prev_measurement_date::timestamp - b.prev2_measurement_date::timestamp)) / 31557600.0 AS delta_years_prev
    FROM base b
),
rates AS (
    SELECT
        d.*,
        CASE
            WHEN d.prev_measurement_date IS NULL OR d.delta_years IS NULL OR d.delta_years <= 0 THEN NULL
            ELSE (d.prev_thickness - d.curr_thickness) / d.delta_years
        END AS corr_rate_mm_per_year,
        CASE
            WHEN d.prev2_measurement_date IS NULL OR d.delta_years_prev IS NULL OR d.delta_years_prev <= 0 THEN NULL
            ELSE (d.prev2_thickness - d.prev_thickness) / d.delta_years_prev
        END AS prev_corr_rate_mm_per_year
    FROM derived d
),
flagged AS (
    SELECT
        r.*,
        CASE
            WHEN r.nominal_eff IS NOT NULL
                AND r.tmin_eff IS NOT NULL
                AND r.prev_thickness IS NOT NULL
                AND r.curr_thickness >= 0.95 * r.nominal_eff
                AND ABS(r.prev_thickness - r.curr_thickness) > (r.nominal_eff - r.tmin_eff) / 2.0
                AND COALESCE(r.prev_corr_rate_mm_per_year, 0) > 0
            THEN TRUE
            ELSE FALSE
        END AS is_replacement
    FROM rates r
),
with_effective_date AS (
    SELECT
        f.*,
        COALESCE(
            MAX(CASE WHEN f.is_replacement THEN f.curr_measurement_date END)
                OVER (PARTITION BY f.equipment_id, f.component_id, f.mms
                      ORDER BY f.curr_measurement_date
                      ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING),
            f.start_date_of_operation
        ) AS effective_start_date
    FROM flagged f
)
SELECT
    measurement_id,
    installation_id,
    installation,
    equipment_id,
    equipment,
    start_date_of_operation,
    component_id,
    component,
    contour,
    mms,
    curr_measurement_date,
    curr_thickness,
    prev_measurement_date,
    prev_thickness,
    prev2_measurement_date,
    prev2_thickness,
    nominal_thickness_mmc,
    nominal_eff,
    tmin_mmc,
    tmin_eff,
    initial_thickness,
    component_type,
    measurement_method_code,
    delta_years,
    delta_years_prev,
    corr_rate_mm_per_year,
    prev_corr_rate_mm_per_year,
    is_replacement,
    effective_start_date,
    CASE
        WHEN effective_start_date IS NOT NULL
        THEN EXTRACT(epoch FROM (curr_measurement_date::timestamp - effective_start_date::timestamp)) / 31557600.0
        ELSE NULL
    END AS equipment_age_years
FROM with_effective_date
WHERE prev_measurement_date IS NOT NULL AND delta_years > 0
WITH DATA;

-- Индексы
CREATE UNIQUE INDEX ux__mv_pipeline_corrosion_base_measurement_id
    ON public._mv_pipeline_corrosion_base (measurement_id);
CREATE INDEX ix__mv_pipeline_corrosion_base_ids
    ON public._mv_pipeline_corrosion_base (installation_id, equipment_id, component_id, mms, curr_measurement_date);
CREATE INDEX ix__mv_pipeline_corrosion_base_contour
    ON public._mv_pipeline_corrosion_base (equipment_id, contour, mms, curr_measurement_date);
CREATE INDEX ix__mv_pipeline_corrosion_base_replacement
    ON public._mv_pipeline_corrosion_base (is_replacement);
