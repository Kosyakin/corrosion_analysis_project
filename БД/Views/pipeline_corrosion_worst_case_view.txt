-- public.pipeline_corrosion_worst_case_view исходный текст

CREATE OR REPLACE VIEW public.pipeline_corrosion_worst_case_view
AS SELECT installation,
    equipment,
    split_part(mms::text, '.'::text, 1) AS section_number,
    measurement_date,
    contour,
    round(max(corrosion_rate_mm_per_year), 2) AS worst_corrosion_rate_mm_per_year,
    min(measurement) AS worst_thickness_measurement,
    water_content,
    h2s_content,
    chlorine_content,
    co2_content,
    oxygen_content,
    hydrochloric_acid_content,
    ammonia_content,
    corrosion_inhibitor_content,
    total_sulfur_compounds,
    total_chlorine_compounds,
    total_acids,
    corrosion_aggressiveness_index,
    h2s_water_ratio,
    h2s_aggressiveness_index,
    total_acidity_index,
    chloride_aggressiveness,
    pitting_corrosion_index,
    underdeposit_corrosion_index,
    stress_corrosion_index,
    corrosion_protection_index,
    operating_temperature,
    operating_pressure,
    material_resistance_score,
    round(equipment_age_years, 0) AS equipment_age_years,
    wall_thickness,
    outer_diameter,
    inner_diameter,
    diameter_to_thickness_ratio,
    cross_sectional_area,
    component_type_id,
    component_type_name,
    nominal_thickness_mmc,
    tmin_mmc,
    material_code,
    material_type,
    thickness_category,
    diameter_category,
    count(*) AS measurement_points_in_section,
    round(avg(corrosion_rate_mm_per_year), 2) AS avg_corrosion_rate_mm_per_year,
    round(min(corrosion_rate_mm_per_year), 2) AS min_corrosion_rate_mm_per_year
   FROM pipeline_corrosion_analysis_new_view
  WHERE is_replaced = false AND corrosion_rate_mm_per_year IS NOT NULL AND corrosion_rate_mm_per_year < 100::numeric AND wall_thickness <= (outer_diameter / 2.0)
  GROUP BY installation, equipment, (split_part(mms::text, '.'::text, 1)), measurement_date, contour, water_content, h2s_content, chlorine_content, co2_content, oxygen_content, hydrochloric_acid_content, ammonia_content, corrosion_inhibitor_content, total_sulfur_compounds, total_chlorine_compounds, total_acids, corrosion_aggressiveness_index, h2s_water_ratio, h2s_aggressiveness_index, total_acidity_index, chloride_aggressiveness, pitting_corrosion_index, underdeposit_corrosion_index, stress_corrosion_index, corrosion_protection_index, operating_temperature, operating_pressure, material_resistance_score, equipment_age_years, wall_thickness, outer_diameter, inner_diameter, diameter_to_thickness_ratio, cross_sectional_area, component_type_id, component_type_name, nominal_thickness_mmc, tmin_mmc, material_code, material_type, thickness_category, diameter_category
 HAVING count(*) >= 1
  ORDER BY installation, equipment, (split_part(mms::text, '.'::text, 1)), measurement_date;
