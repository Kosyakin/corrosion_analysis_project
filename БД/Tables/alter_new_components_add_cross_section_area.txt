-- Add cross-sectional area (metal) to new_components
-- A = π/4 * (OD^2 - ID^2); if ID is NULL, approximate as (OD - 2*initial_thickness)

ALTER TABLE public.new_components
ADD COLUMN IF NOT EXISTS cross_section_area_mm2 numeric(18, 4)
	GENERATED ALWAYS AS (
		CASE
			WHEN outer_diameter IS NOT NULL THEN
				(
					pi()::numeric * (
						outer_diameter * outer_diameter
						- COALESCE(
							inner_diameter,
							CASE
								WHEN initial_thickness IS NOT NULL THEN GREATEST(outer_diameter - 2 * initial_thickness, 0)
								ELSE NULL
							END
						) * COALESCE(
							inner_diameter,
							CASE
								WHEN initial_thickness IS NOT NULL THEN GREATEST(outer_diameter - 2 * initial_thickness, 0)
								ELSE NULL
							END
						)
					)
				) / 4
			ELSE NULL
		END
	) STORED;

CREATE INDEX IF NOT EXISTS ix_new_components_cross_section_area
	ON public.new_components (cross_section_area_mm2);
