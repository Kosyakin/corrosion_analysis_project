CREATE TABLE public.equipment_components (
	id serial4 NOT NULL,
	installation varchar(50) NULL,
	equipment varchar(100) NULL,
	component varchar(100) NULL,
	source_component varchar(100) NULL,
	material_code varchar(50) NULL,
	component_type varchar(50) NULL,
	operating_temperature numeric(6, 1) NULL,
	operating_pressure numeric(8, 3) NULL,
	equipment_type_tm varchar(50) NULL,
	outer_diameter numeric(8, 1) NULL,
	inner_diameter numeric(8, 1) NULL,
	initial_thickness numeric(6, 1) NULL,
	tank_volume numeric(10, 1) NULL,
	material_type varchar(50) NULL,
	CONSTRAINT equipment_components_pkey PRIMARY KEY (id)
);
